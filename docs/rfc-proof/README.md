From final RFC5036 checkout:

```sh
corepack pnpm --filter @module-federation/rspack run test
corepack pnpm --filter @module-federation/sdk run test
corepack pnpm --filter @module-federation/webpack-bundler-runtime run test
corepack pnpm exec prettier --check .
node tools/scripts/prove-runtime-selection-slot.cjs
node tools/scripts/prove-runtime-node-payload.cjs
```

```sh
corepack pnpm --filter @module-federation/runtime-core run test
```
